      ```swift
      func runSonosMeasureLatency(args: [String]) async {
          // --- arg parsing ---
          guard let ip = args.first, !ip.hasPrefix("--") else {
              print("❌ Usage: sonos measure-latency <ip> [--trials N] [--start-relay] [--mac-ip <ip>]")
              exit(1)
          }
          var trials = 5
          var startOwnRelay = false
          var macIPOverride: String? = nil
          var i = 1
          while i < args.count {
              switch args[i] {
              case "--trials":
                  if i + 1 < args.count, let n = Int(args[i + 1]) {
                      trials = max(3, min(10, n))
                      i += 2
                  } else { print("❌ --trials requires integer 3..10"); exit(1) }
              case "--start-relay":
                  startOwnRelay = true
                  i += 1
              case "--mac-ip":
                  if i + 1 < args.count {
                      macIPOverride = args[i + 1]
                      i += 2
                  } else { print("❌ --mac-ip requires an IP"); exit(1) }
              default:
                  i += 1
              }
          }

          print("⏱  SPIKE-05 — Pause-to-silence latency, \(trials) trials")
          print("   Target: ≤2000ms (aspirational); AirConnect baseline: 8000–10000ms")
          print("   Methodology: manual stopwatch via Enter-key (per RESEARCH.md Section 6)")
          print("   Sonos: \(ip)")
          print("")
          print("⚠️  Note: if you abort mid-run, the Sonos may be left in PAUSED state.")
          print("   Recover via the Sonos app, or by running: swift run api-explorer sonos play \(ip)")
          print("")

          // --- optional own-relay bootstrap ---
          var listener: NWListener? = nil
          if startOwnRelay {
              let macIP = macIPOverride ?? detectMacLANIP() ?? ""
              guard !macIP.isEmpty else {
                  print("❌ Could not detect Mac LAN IP; pass --mac-ip <ip>")
                  exit(1)
              }
              let fixturePath = URL(fileURLWithPath: ".planning/phases/01-spike/fixtures/test-stream.wav")
              guard FileManager.default.fileExists(atPath: fixturePath.path) else {
                  print("❌ Missing fixture at \(fixturePath.path) — run PLAN 01 first")
                  exit(1)
              }
              do {
                  // Positional unlabeled binding against Plan 01-03's labeled return tuple
                  // (listener: NWListener, port: UInt16, sonosURIString: String, fileSuffix: String).
                  // Swift accepts unlabeled bindings against labeled tuples — the inner `relayListener` name
                  // avoids shadowing the outer `var listener: NWListener?` (which the `defer` block uses for cleanup).
                  let (relayListener, port, sonosURI, fileSuffix) = try await serveStaticAudioFile(
                      path: fixturePath, contentType: "audio/wav", macIP: macIP, port: 0
                  )
                  _ = fileSuffix  // unused at this call-site; bound for tuple-arity completeness
                  listener = relayListener
                  print("📻 Started own relay: \(sonosURI) on port \(port)")
                  // SetAVTransportURI + Play
                  let setBody = """
                  <u:SetAVTransportURI xmlns:u="urn:schemas-upnp-org:service:AVTransport:1">
                    <InstanceID>0</InstanceID>
                    <CurrentURI>\(sonosURI)</CurrentURI>
                    <CurrentURIMetaData></CurrentURIMetaData>
                  </u:SetAVTransportURI>
                  """
                  _ = try await sendSonosSOAP(host: ip, service: .avTransport, action: "SetAVTransportURI", bodyXML: setBody)
                  let playBody = """
                  <u:Play xmlns:u="urn:schemas-upnp-org:service:AVTransport:1">
                    <InstanceID>0</InstanceID><Speed>1</Speed>
                  </u:Play>
                  """
                  _ = try await sendSonosSOAP(host: ip, service: .avTransport, action: "Play", bodyXML: playBody)
                  // Wait a bit for buffer to fill
                  print("   Waiting 8s for Sonos input buffer to fill...")
                  try await Task.sleep(for: .seconds(8))
              } catch {
                  print("❌ Failed to start relay: \(error)")
                  exit(1)
              }
          } else {
              print("Pre-flight: ensure a Sonos zone at \(ip) is currently playing via the spike.")
              print("            (Run `swift run api-explorer sonos play \(ip)` in another terminal,")
              print("             OR re-run with --start-relay to have this command bootstrap a relay.)")
              print("Press Enter when audio is audibly playing on the Sonos...")
              _ = readLine()
          }

          // --- ensure cleanup at exit ---
          defer {
              listener?.cancel()
              // Stop SOAP issued asynchronously; can't await in defer for non-async closure,
              // so use a Task and let the process exit catch it.
              Task { @Sendable in
                  let stopBody = """
                  <u:Stop xmlns:u="urn:schemas-upnp-org:service:AVTransport:1">
                    <InstanceID>0</InstanceID>
                  </u:Stop>
                  """
                  _ = try? await sendSonosSOAP(host: ip, service: .avTransport, action: "Stop", bodyXML: stopBody)
              }
          }

          // --- trial loop ---
          var latenciesMs: [Int] = []
          let pauseBody = """
          <u:Pause xmlns:u="urn:schemas-upnp-org:service:AVTransport:1">
            <InstanceID>0</InstanceID>
          </u:Pause>
          """
          let playBody = """
          <u:Play xmlns:u="urn:schemas-upnp-org:service:AVTransport:1">
            <InstanceID>0</InstanceID><Speed>1</Speed>
          </u:Play>
          """

          for trial in 1...trials {
              print("")
              print("Trial \(trial)/\(trials):")
              print("  → Press Enter to send Pause SOAP. Then press Enter AGAIN the moment")
              print("    you hear silence at the Sonos. (You're stopwatching from finger to ear.)")
              _ = readLine()
              let pauseSent = Date()
              do {
                  _ = try await sendSonosSOAP(host: ip, service: .avTransport, action: "Pause", bodyXML: pauseBody)
              } catch {
                  print("  ⚠️  Pause SOAP failed: \(error). Skipping trial.")
                  continue
              }
              print("  ✓ Pause SOAP sent. Listening...")
              _ = readLine()
              let silenceHeard = Date()
              let latencyMs = Int(silenceHeard.timeIntervalSince(pauseSent) * 1000)
              latenciesMs.append(latencyMs)
              print("  ⏱  Trial \(trial): \(latencyMs)ms")

              // Resume for the next trial
              do {
                  _ = try await sendSonosSOAP(host: ip, service: .avTransport, action: "Play", bodyXML: playBody)
                  print("  ▶ Play resumed; waiting 4s for buffer to refill before next trial...")
                  try await Task.sleep(for: .seconds(4))
              } catch {
                  print("  ⚠️  Play resume failed: \(error). Trial \(trial+1) may not have audio.")
              }
          }

          // --- compute + report ---
          guard !latenciesMs.isEmpty else {
              print("\n❌ NO-GO: zero successful trials (every Pause SOAP failed)")
              exit(1)
          }

          let sorted = latenciesMs.sorted()
          let median = sorted[sorted.count / 2]
          let mn = sorted.first ?? 0
          let mx = sorted.last ?? 0

          print("")
          print("📊 Results:")
          print("   Trials: \(latenciesMs)")
          print("   Median: \(median)ms")
          print("   Min:    \(mn)ms")
          print("   Max:    \(mx)ms")

          let verdict: String
          let exitCode: Int32
          if median <= 2000 {
              verdict = "GO"; exitCode = 0
              print("   ✅ GO: median ≤ 2.0s — use SOAP Pause for user pause; SOAP Play for resume")
          } else if median <= 2500 {
              verdict = "GO-WITH-WORKAROUND"; exitCode = 2
              print("   ⚠️  GO-WITH-WORKAROUND: median ≤ 2.5s")
              print("      Recommend Pause→Stop substitution: SOAP Stop drains buffer faster")
              print("      (adds ~500ms to resume but cuts pause-to-silence in half).")
              print("      Document the tradeoff in PlayerService UX.")
          } else {
              verdict = "NO-GO"; exitCode = 1
              print("   ❌ NO-GO: median > 2.5s — pause UX needs a 'will pause shortly' indicator")
          }

          // --- persist results JSON for PLAN 06 ingestion ---
          let results = LatencyResults(
              trials: latenciesMs,
              medianMs: median,
              minMs: mn,
              maxMs: mx,
              verdict: verdict,
              timestamp: ISO8601DateFormatter().string(from: Date()),
              methodology: "manual-stopwatch-\(trials)-trials",
              exitCode: Int(exitCode)
          )
          let outURL = URL(fileURLWithPath: ".planning/phases/01-spike/fixtures/latency-results.json")
          do {
              let encoder = JSONEncoder()
              encoder.outputFormatting = [.prettyPrinted, .sortedKeys]
              let data = try encoder.encode(results)
              try data.write(to: outURL)
              print("   📝 Results written to \(outURL.path)")
          } catch {
              print("   ⚠️  Failed to write results JSON: \(error)")
          }

          exit(exitCode)
      }
      ```
