      ```swift
      #if SPIKE_TAP_COEXIST
      /// Single-producer (audio thread) / single-consumer (CLI drainer) fixed-size ring of Float samples.
      /// Per-channel storage; 0.5 s capacity at 48 kHz stereo Float32 (~192 KB total).
      /// Pre-allocated; never resized. Audio-thread `tryWrite` returns false on overflow → caller drops the frame.
      final class SpikeFanoutRing: @unchecked Sendable {
          let capacityFrames: Int
          let channels: Int
          private let storage: UnsafeMutableBufferPointer<Float>  // channels × capacityFrames, channel-major
          // Swift 6 strict concurrency: Atomic stored properties on non-actor classes require nonisolated(unsafe).
          // The atomic operations themselves provide the memory-safety guarantee; the annotation only tells the
          // compiler we've audited the cross-isolation access pattern (single-producer audio thread + single-consumer CLI).
          private nonisolated(unsafe) var writeIndex = Atomic<UInt64>(0)
          private nonisolated(unsafe) var readIndex = Atomic<UInt64>(0)

          init(capacityFrames: Int, channels: Int) {
              self.capacityFrames = capacityFrames
              self.channels = channels
              self.storage = UnsafeMutableBufferPointer.allocate(capacity: channels * capacityFrames)
              self.storage.initialize(repeating: 0.0)
          }

          deinit {
              self.storage.deinitialize()
              self.storage.deallocate()
          }

          /// RT-safe — call from audio thread.
          /// Returns the number of frames actually written (may be less than `count` on overflow).
          func tryWrite(channel: Int, src: UnsafePointer<Float>, count: Int) -> Int {
              let w = writeIndex.load(ordering: .acquiring)
              let r = readIndex.load(ordering: .acquiring)
              let available = capacityFrames - Int(w &- r)
              let toWrite = min(count, available)
              if toWrite <= 0 { return 0 }
              for i in 0..<toWrite {
                  let pos = (Int(w) &+ i) % capacityFrames
                  storage[channel * capacityFrames + pos] = src[i]
              }
              if channel == channels - 1 {
                  // Only advance writeIndex after the LAST channel of this frame batch is written
                  writeIndex.store(w &+ UInt64(toWrite), ordering: .releasing)
              }
              return toWrite
          }

          /// Called from CLI thread (single consumer).
          /// Returns frames actually read into `dst[channel]` (may be less than `count`).
          func read(into dstPerChannel: [UnsafeMutablePointer<Float>], maxFrames: Int) -> Int {
              let w = writeIndex.load(ordering: .acquiring)
              let r = readIndex.load(ordering: .acquiring)
              let available = Int(w &- r)
              let toRead = min(maxFrames, available)
              if toRead <= 0 { return 0 }
              for ch in 0..<channels {
                  for i in 0..<toRead {
                      let pos = (Int(r) &+ i) % capacityFrames
                      dstPerChannel[ch][i] = storage[ch * capacityFrames + pos]
                  }
              }
              readIndex.store(r &+ UInt64(toRead), ordering: .releasing)
              return toRead
          }
      }
      #endif
      ```

      func fileSize(at url: URL) -> Int {
          (try? FileManager.default.attributesOfItem(atPath: url.path)[.size] as? Int) ?? 0
      }

      // Minimal 16-bit PCM WAV writer (44-byte header + samples).
      // Float32 input → clamped Int16 output. Hand-rolled per RESEARCH.md "Don't Hand-Roll"
      // exception (44 well-known bytes, simpler than AVAudioFile dance for the spike).
      final class SpikeWAVWriter {
          private var handle: FileHandle?
          private var dataBytesWritten: UInt32 = 0
          private var sampleRate: UInt32 = 48_000
          private var channels: UInt16 = 2
          private var bitsPerSample: UInt16 = 16
          private var fileURL: URL?

          func open(at url: URL, sampleRate: UInt32, channels: UInt16, bitsPerSample: UInt16) throws {
              self.sampleRate = sampleRate
              self.channels = channels
              self.bitsPerSample = bitsPerSample
              self.fileURL = url
              FileManager.default.createFile(atPath: url.path, contents: nil)
              self.handle = try FileHandle(forWritingTo: url)
              // Write placeholder 44-byte header; will be patched in close()
              try self.handle?.write(contentsOf: Data(repeating: 0, count: 44))
          }

          func write(channel0: UnsafePointer<Float>, channel1: UnsafePointer<Float>, frames: Int) {
              guard let handle = self.handle else { return }
              var interleaved = [Int16](repeating: 0, count: frames * Int(self.channels))
              for i in 0..<frames {
                  let l = max(-1.0, min(1.0, channel0[i]))
                  let r = max(-1.0, min(1.0, channel1[i]))
                  interleaved[i * 2] = Int16(l * 32767.0)
                  interleaved[i * 2 + 1] = Int16(r * 32767.0)
              }
              let bytes = interleaved.withUnsafeBufferPointer { buf -> Data in
                  Data(buffer: buf)
              }
              try? handle.write(contentsOf: bytes)
              self.dataBytesWritten &+= UInt32(bytes.count)
          }

          func close() throws {
              guard let handle = self.handle, let url = self.fileURL else { return }
              try handle.seek(toOffset: 0)
              try handle.write(contentsOf: self.makeHeader())
              try handle.close()
              self.handle = nil
              _ = url
          }

          private func makeHeader() -> Data {
              let totalSize = UInt32(36) &+ self.dataBytesWritten
              let byteRate = self.sampleRate * UInt32(self.channels) * UInt32(self.bitsPerSample) / 8
              let blockAlign = self.channels * (self.bitsPerSample / 8)
              var d = Data()
              d.append("RIFF".data(using: .ascii) ?? Data())
              d.append(le32(totalSize))
              d.append("WAVEfmt ".data(using: .ascii) ?? Data())
              d.append(le32(16))         // fmt chunk size
              d.append(le16(1))          // PCM
              d.append(le16(self.channels))
              d.append(le32(self.sampleRate))
              d.append(le32(byteRate))
              d.append(le16(blockAlign))
              d.append(le16(self.bitsPerSample))
              d.append("data".data(using: .ascii) ?? Data())
              d.append(le32(self.dataBytesWritten))
              return d
          }
          private func le16(_ v: UInt16) -> Data {
              var x = v.littleEndian; return Data(bytes: &x, count: 2)
          }
          private func le32(_ v: UInt32) -> Data {
              var x = v.littleEndian; return Data(bytes: &x, count: 4)
          }
      }
      #endif
