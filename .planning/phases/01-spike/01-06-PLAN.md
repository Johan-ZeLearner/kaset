      ```swift
      func runSonosSpikeAll(args: [String]) async {
          guard let ip = args.first, !ip.hasPrefix("--") else {
              print("❌ Usage: sonos spike-all <ip> [--skip-tap-coexist] [--skip-discover] [--trials N]")
              exit(1)
          }
          var skipTap = false
          var skipDiscover = false
          var trials = "5"
          var i = 1
          while i < args.count {
              switch args[i] {
              case "--skip-tap-coexist": skipTap = true; i += 1
              case "--skip-discover":    skipDiscover = true; i += 1
              case "--trials":
                  if i + 1 < args.count { trials = args[i + 1]; i += 2 } else { print("❌ --trials requires N"); exit(1) }
              default: i += 1
              }
          }

          struct BetResult {
              let name: String
              let cmd: [String]
              let exitCode: Int32
              let durationSec: Double
          }

          func runBet(name: String, cliArgs: [String]) async -> BetResult {
              print("")
              print("════════════════════════════════════════════════════════════")
              print("  \(name)")
              print("  swift run api-explorer \(cliArgs.joined(separator: " "))")
              print("════════════════════════════════════════════════════════════")
              let startTime = Date()
              let process = Process()
              process.executableURL = URL(fileURLWithPath: "/usr/bin/env")
              process.arguments = ["swift", "run", "api-explorer"] + cliArgs
              do {
                  try process.run()
              } catch {
                  print("❌ Failed to launch subprocess: \(error)")
                  return BetResult(name: name, cmd: cliArgs, exitCode: 127, durationSec: 0)
              }
              await withCheckedContinuation { (cont: CheckedContinuation<Void, Never>) in
                  process.terminationHandler = { _ in cont.resume() }
              }
              let duration = Date().timeIntervalSince(startTime)
              return BetResult(name: name, cmd: cliArgs, exitCode: process.terminationStatus, durationSec: duration)
          }

          var results: [BetResult] = []

          if !skipDiscover {
              results.append(await runBet(name: "SPIKE-04 — SSDP discovery", cliArgs: ["sonos", "discover", "--verbose"]))
          }

          results.append(await runBet(name: "SPIKE-03 — Encoder + HTTP relay", cliArgs: ["sonos", "play", ip, "--format", "both"]))

          if !skipTap {
              results.append(await runBet(name: "SPIKE-02 — Dual-consumer tap", cliArgs: ["sonos", "tap-coexist"]))
          } else {
              print("\n⏭  Skipping SPIKE-02 (--skip-tap-coexist)")
          }

          results.append(await runBet(name: "SPIKE-05 — Pause-to-silence latency", cliArgs: ["sonos", "measure-latency", ip, "--trials", trials]))

          // --- aggregate ---
          print("")
          print("════════════════════════════════════════════════════════════")
          print("  SPIKE-ALL SUMMARY")
          print("════════════════════════════════════════════════════════════")
          for r in results {
              let verdict: String
              switch r.exitCode {
              case 0: verdict = "✅ GO"
              case 2: verdict = "⚠️  GO-WITH-WORKAROUND"
              default: verdict = "❌ NO-GO (exit \(r.exitCode))"
              }
              print(String(format: "  %-44s %@  (%.0fs)", r.name, verdict, r.durationSec))
          }
          print("")
          print("Next step: review per-plan SUMMARY.md files, redact LAN IPs/USNs, and")
          print("           generate .planning/phases/01-spike/REPORT.md per RESEARCH.md Section 7 template.")
          print("           See PLAN 06 task 2 for the report-generation steps.")

          // Exit code = max severity across bets (0 = all GO, 2 = at least one GO-WITH-WORKAROUND, 1 = at least one NO-GO)
          var aggregate: Int32 = 0
          for r in results {
              if r.exitCode == 1 { aggregate = 1 } // NO-GO dominates
              else if r.exitCode == 2 && aggregate == 0 { aggregate = 2 }
          }
          exit(aggregate)
      }
      ```
