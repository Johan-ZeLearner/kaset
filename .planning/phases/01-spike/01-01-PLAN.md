      ```swift
      func sendSonosSOAP(
          host: String,
          service: SonosService,
          action: String,
          bodyXML: String
      ) async throws -> Data {
          let endpoint = service == .avTransport
              ? "MediaRenderer/AVTransport/Control"
              : "MediaRenderer/RenderingControl/Control"
          let serviceURN = service == .avTransport
              ? "urn:schemas-upnp-org:service:AVTransport:1"
              : "urn:schemas-upnp-org:service:RenderingControl:1"
          guard let url = URL(string: "http://\(host):1400/\(endpoint)") else {
              throw SonosSpikeError.invalidURL("http://\(host):1400/\(endpoint)")
          }
          let envelope = """
          <?xml version="1.0" encoding="utf-8"?>
          <s:Envelope xmlns:s="http://schemas.xmlsoap.org/soap/envelope/" s:encodingStyle="http://schemas.xmlsoap.org/soap/encoding/">
            <s:Body>
          \(bodyXML)
            </s:Body>
          </s:Envelope>
          """
          var request = URLRequest(url: url)
          request.httpMethod = "POST"
          request.timeoutInterval = 3.0
          request.setValue("text/xml; charset=\"utf-8\"", forHTTPHeaderField: "Content-Type")
          request.setValue("\"\(serviceURN)#\(action)\"", forHTTPHeaderField: "SOAPACTION")
          request.setValue("close", forHTTPHeaderField: "Connection")
          request.httpBody = envelope.data(using: .utf8)
          let (data, response) = try await URLSession.shared.data(for: request)
          guard let http = response as? HTTPURLResponse else {
              throw SonosSpikeError.malformedResponse
          }
          // Per RESEARCH.md Pitfall 1: SOAP encodes errors at body level, not HTTP status
          if let bodyString = String(data: data, encoding: .utf8),
             bodyString.contains("<s:Fault>") || bodyString.contains("<errorCode>") {
              throw SonosSpikeError.soapFault(String(bodyString.prefix(500)))
          }
          guard http.statusCode == 200 else {
              throw SonosSpikeError.httpStatus(http.statusCode)
          }
          return data
      }
      ```

    (4) Add the LAN IP detector per RESEARCH.md Section 5 hint ("Use `getifaddrs()` to enumerate non-loopback IPv4 addresses; pick the one matching the network the user's Sonos is on"):
      ```swift
      func detectMacLANIP() -> String? {
          var ifaddrPtr: UnsafeMutablePointer<ifaddrs>?
          guard getifaddrs(&ifaddrPtr) == 0, let firstAddr = ifaddrPtr else { return nil }
          defer { freeifaddrs(ifaddrPtr) }
          var candidates: [String] = []
          var ptr: UnsafeMutablePointer<ifaddrs>? = firstAddr
          while let current = ptr {
              defer { ptr = current.pointee.ifa_next }
              guard let addr = current.pointee.ifa_addr else { continue }
              guard addr.pointee.sa_family == sa_family_t(AF_INET) else { continue }
              let name = String(cString: current.pointee.ifa_name)
              // Skip loopback, awdl, utun, llw — keep only en0/en1/etc.
              guard name.hasPrefix("en") else { continue }
              var hostBuf = [CChar](repeating: 0, count: Int(NI_MAXHOST))
              let result = getnameinfo(
                  addr,
                  socklen_t(addr.pointee.sa_len),
                  &hostBuf,
                  socklen_t(hostBuf.count),
                  nil, 0,
                  NI_NUMERICHOST
              )
              if result == 0 {
                  let ip = String(cString: hostBuf)
                  // Heuristic: prefer RFC-1918 LAN ranges
                  if ip.hasPrefix("192.168.") || ip.hasPrefix("10.") || ip.hasPrefix("172.") {
                      candidates.append(ip)
                  }
              }
          }
          return candidates.first
      }
      ```
